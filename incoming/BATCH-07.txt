# BATCH 07 - Public Form to Intake API Wiring / Local API Boot Proof

## Purpose

This batch connects the public Free Destiny Map form shell toward the safe intake API boundary created in BATCH 05 and the local persistence shell created in BATCH 06.

It does not claim production readiness.

## What this batch creates

- public form client adapter
- connected Free Destiny Map form component
- updated Free Destiny Map page
- public API wiring contract
- local API boot proof script
- runtime proof report
- validator and validation report

## Runtime boundary

The connected form posts to:

`/api/public/free-destiny-map/intake/local`

That endpoint remains a local runtime shell.

The backend decides whether local JSONL writing is enabled through:

`KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE`

If the flag is not enabled, the API still returns safe shell feedback without writing data.

## What this batch does not do yet

- no production database storage
- no Supabase service role usage
- no email sending
- no payment processing
- no document upload
- no official application submission
- no Oracle runtime claim
- no production readiness claim

## Safety

The public form must only capture basic route-mapping information and consent.

It must not capture raw ID numbers, passwords, portal logins, card details, CVV, document files, base64 documents, or official application credentials.

## Oracle Lite impact

none_runtime_yet.

This batch creates UI wiring, client adapter, local boot proof, validator, and reports only.

## Next action

After validation, BATCH 08 should create local API boot/server wiring or the first developer runtime command that proves the route can be exercised locally.

export const FREE_DESTINY_MAP_PUBLIC_ROUTE = "/free-destiny-map" as const;

export const FREE_DESTINY_MAP_LOCAL_INTAKE_ENDPOINT = "/api/public/free-destiny-map/intake/local" as const;

export const FREE_DESTINY_MAP_LOCAL_WRITE_FLAG = "KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE" as const;

export const FREE_DESTINY_MAP_API_WIRING_NOTICE =
  "This public form wiring targets a local intake shell. It is not production storage, not payment, not email delivery, not document upload, and not official application submission.";

export interface FreeDestinyMapClientSubmission {
  fullName?: string;
  province?: string;
  currentSituation?: string;
  knownRoute?: string;
  routeInterests?: string[];
  needsAccommodation?: boolean;
  accommodationArea?: string;
  needsFundingHelp?: boolean;
  needsCvHelp?: boolean;
  hasDocumentsReady?: boolean;
  randomSourceName?: string;
  consentToBeContacted?: boolean;
  consentToProcessBasicInfo?: boolean;
}

export interface FreeDestinyMapClientResponse {
  accepted: boolean;
  reference: string;
  productionReady: false;
  localPersistenceEnabled?: boolean;
  localWriteAttempted?: boolean;
  localWriteSucceeded?: boolean;
  storageMode?: "disabled" | "local_jsonl_shell";
  safeNotice: string;
  nextAction: string;
  issues: Array<{
    field: string;
    code: string;
    message: string;
  }>;
}

import {
  FREE_DESTINY_MAP_LOCAL_INTAKE_ENDPOINT,
  type FreeDestinyMapClientResponse,
  type FreeDestinyMapClientSubmission
} from "../../../../../packages/contracts/src/free-destiny-map-api-wiring";

export function buildFreeDestinyMapClientPayload(formData: FormData): FreeDestinyMapClientSubmission {
  const routeInterests = formData
    .getAll("routeInterests")
    .map((item) => String(item))
    .filter(Boolean);

  return {
    fullName: String(formData.get("fullName") ?? "").trim(),
    province: String(formData.get("province") ?? "").trim(),
    currentSituation: String(formData.get("currentSituation") ?? "").trim(),
    knownRoute: String(formData.get("knownRoute") ?? "").trim(),
    routeInterests,
    needsAccommodation: formData.get("needsAccommodation") === "on",
    accommodationArea: String(formData.get("accommodationArea") ?? "").trim(),
    needsFundingHelp: formData.get("needsFundingHelp") === "on",
    needsCvHelp: formData.get("needsCvHelp") === "on",
    hasDocumentsReady: formData.get("hasDocumentsReady") === "on",
    randomSourceName: String(formData.get("randomSourceName") ?? "").trim(),
    consentToBeContacted: formData.get("consentToBeContacted") === "on",
    consentToProcessBasicInfo: formData.get("consentToProcessBasicInfo") === "on"
  };
}

export async function submitFreeDestinyMapClientPayload(
  payload: FreeDestinyMapClientSubmission
): Promise<FreeDestinyMapClientResponse> {
  const response = await fetch(FREE_DESTINY_MAP_LOCAL_INTAKE_ENDPOINT, {
    method: "POST",
    headers: {
      "content-type": "application/json"
    },
    body: JSON.stringify(payload)
  });

  const data = (await response.json()) as FreeDestinyMapClientResponse;

  return data;
}

#!/usr/bin/env node

import path from "node:path";

];

function exists(rel) {
  return fs.existsSync(path.join(root, rel));
}

function read(rel) {
  return fs.readFileSync(path.join(root, rel), "utf8").replace(/^\uFEFF/, "");
}

function parseJson(rel) {
  return JSON.parse(read(rel));
}

const failures = [];
const warnings = [];
const checked = [];

for (const file of requiredFiles) {
  checked.push(file);
  if (!exists(file)) failures.push(`Missing required file: ${file}`);
  else if (fs.statSync(path.join(root, file)).size === 0) failures.push(`Empty required file: ${file}`);
}

if (exists("manifests/runtime/free-destiny-map-api-wiring-shell.json")) {
  const manifest = parseJson("manifests/runtime/free-destiny-map-api-wiring-shell.json");
  if (manifest.productionReady !== false) failures.push("Manifest must keep productionReady false.");
  if (manifest.apiEndpoint !== "/api/public/free-destiny-map/intake/local") failures.push("Manifest endpoint mismatch.");
  if (manifest.storesProductionData !== false) failures.push("Manifest must declare storesProductionData false.");
  if (manifest.takesPayment !== false) failures.push("Manifest must declare takesPayment false.");
  if (manifest.sendsEmail !== false) failures.push("Manifest must declare sendsEmail false.");
  if (manifest.collectsDocuments !== false) failures.push("Manifest must declare collectsDocuments false.");
  if (!String(manifest.oracleLiteImpact ?? "").includes("none_runtime_yet")) failures.push("Manifest must declare none_runtime_yet.");
}

const tokenChecks = [
  ["packages/contracts/src/free-destiny-map-api-wiring.ts", ["FREE_DESTINY_MAP_LOCAL_INTAKE_ENDPOINT", "FREE_DESTINY_MAP_API_WIRING_NOTICE", "FreeDestinyMapClientSubmission", "FreeDestinyMapClientResponse"]],

  ["apps/web/src/app/free-destiny-map/page.tsx", ["FreeDestinyMapApiForm"]],
  ["scripts/proof/prove-free-destiny-map-api-wiring.mjs", ["RUNTIME_SHELL_PROVEN", "client-posts-local-endpoint", "route-local-endpoint-exists"]]
];

for (const [file, tokens] of tokenChecks) {
  if (!exists(file)) continue;
  const content = read(file);
  for (const token of tokens) {
    if (!content.includes(token)) failures.push(`${file} missing token: ${token}`);
  }
}

if (exists("reports/runtime/free-destiny-map-api-wiring-proof.json")) {
  const proof = parseJson("reports/runtime/free-destiny-map-api-wiring-proof.json");
  if (proof.passed !== true) failures.push("Runtime proof report must pass.");
  if (proof.status !== "RUNTIME_SHELL_PROVEN") failures.push("Runtime proof status must be RUNTIME_SHELL_PROVEN.");
}

const forbiddenRuntimePatterns = [
  [/\bidentityNumber\s*[:=]/i, "identity number field assignment"],
  [/\bidNumber\s*[:=]/i, "id number field assignment"],
  [/\bpassportNumber\s*[:=]/i, "passport number field assignment"],
  [/\bpassword\s*[:=]/i, "password field assignment"],
  [/\bcardNumber\s*[:=]/i, "card number field assignment"],
  [/\bcvv\s*[:=]/i, "cvv field assignment"],
  [/\bdocumentBase64\s*[:=]/i, "document payload assignment"],
  [/officialApplicationSubmitted:\s*true/i, "official application submitted true"],
  [/paymentInstructionSent:\s*true/i, "payment instruction sent true"],
  [/emailSent:\s*true/i, "email sent true"],
  [/productionReady\s*[:=]\s*true/i, "productionReady true"]
];

const filesToScan = requiredFiles
  .filter((item) => exists(item))
  .filter((item) => !item.includes("validate-free-destiny-map-api-wiring.mjs"))
  .filter((item) => !item.includes("prove-free-destiny-map-api-wiring.mjs"));

for (const file of filesToScan) {
  const content = read(file);
  for (const [pattern, label] of forbiddenRuntimePatterns) {
    if (pattern.test(content)) failures.push(`${file} contains forbidden runtime pattern: ${label}`);
  }
}

const passed = failures.length === 0;

const report = {
  schema: "kingsmen.validation-report.v1",
  module: "BATCH 07 - Public Form to Intake API Wiring / Local API Boot Proof",
  status: passed ? "VALIDATED" : "BUILT_PENDING_VALIDATION",
  passed,
  generated_at: new Date().toISOString(),
  files_expected: requiredFiles,
  files_checked: checked.length,
  missing_files: requiredFiles.filter((file) => !exists(file)),
  failed_checks: failures,
  warnings,
  oracle_lite_impact: "none_runtime_yet",
