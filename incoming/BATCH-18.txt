# BATCH 18 - Controlled Next-to-Backend Bridge Proof

## Purpose

This batch proves the first controlled bridge from the real Next local API route into the separate backend runtime boundary.

BATCH 15B proved the same-origin local Next web plus Next API route.

BATCH 16 proved the controlled production build.

BATCH 17 proved the separate backend local HTTP service boundary.

BATCH 18 now connects the proven Next API route to the proven backend boundary using explicit local environment flags.

## What this batch proves

- BATCH 17 backend boundary proof exists and passed
- BATCH 16 production build proof exists and passed
- Next dev server can boot locally
- backend boundary server can boot locally
- `/free-destiny-map` returns HTTP 200 from Next
- `/api/public/free-destiny-map/intake/local` returns HTTP 202 from Next
- Next API route calls backend boundary
- backend boundary returns HTTP 202
- backend boundary writes local JSONL proof
- bridge response confirms backend boundary was reached
- production claims remain blocked
- official submission, payment, document upload and email remain blocked
- both servers shut down after proof

## What this batch does not claim

- no Oracle deployment
- no Docker Compose boot
- no PostgreSQL runtime
- no Redis queue
- no public production URL
- no production backend URL
- no email sending
- no payment processing
- no document upload
- no official application submission

## Oracle Lite impact

none_runtime_yet.

This is a controlled local bridge proof only.

## Next action

After validation, BATCH 19 should prepare Oracle Lite container/runtime boundaries or create a controlled Docker-local web plus backend compose proof.

import fs from "node:fs";
import path from "node:path";
import { NextRequest, NextResponse } from "next/server";

const endpoint = "/api/public/free-destiny-map/intake/local";
const localWriteFlagName = "KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE";
const backendBridgeFlagName = "KINGSMEN_ENABLE_BACKEND_BRIDGE";
const backendBaseUrlEnvName = "KINGSMEN_BACKEND_BASE_URL";

type LocalIssue = {
  field: string;
  code: string;
  message: string;
};

type BackendBridgeResult = {
  attempted: boolean;
  succeeded: boolean;
  status: number;
  data: Record<string, unknown> | null;
  error: string | null;
};

function hasSafeConsent(payload: Record<string, unknown>): boolean {
  return payload.consentToBeContacted === true && payload.consentToProcessBasicInfo === true;
}

function hasRouteIntent(payload: Record<string, unknown>): boolean {
  return (
    (Array.isArray(payload.routeInterests) && payload.routeInterests.length > 0) ||
    Boolean(payload.currentSituation) ||
    Boolean(payload.knownRoute) ||
    Boolean(payload.randomSourceName)
  );
}

function sensitiveKeys(payload: Record<string, unknown>): string[] {
  return Object.keys(payload).filter((key) =>
    /identity|idNumber|passport|password|card|cvv|documentBase64/i.test(key)
  );
}

function buildLocalValidationIssues(payload: Record<string, unknown>): LocalIssue[] {
  const issues: LocalIssue[] = [];

  if (!hasSafeConsent(payload)) {
    issues.push({
      field: "consent",
      code: "CONSENT_REQUIRED",
      message: "Contact and basic-info processing consent are required for the local Free Destiny Map intake shell."
    });
  }

  if (!hasRouteIntent(payload)) {
    issues.push({
      field: "intent",
      code: "INTENT_REQUIRED",
      message: "At least one route interest, current situation, known route, or source-check item is required."
    });
  }

  const blockedSensitiveKeys = sensitiveKeys(payload);

  if (blockedSensitiveKeys.length > 0) {
    issues.push({
      field: "payload",
      code: "SENSITIVE_FIELDS_BLOCKED",
      message: "Sensitive fields are blocked in the local Free Destiny Map intake shell."
    });
  }

  return issues;
}

function normalizedBackendBaseUrl(): string | null {
  const value = process.env[backendBaseUrlEnvName];

  if (!value) {
    return null;
  }

  return value.replace(/\/+$/, "");
}

async function postToBackendBoundary(payload: Record<string, unknown>): Promise<BackendBridgeResult> {
  const backendBaseUrl = normalizedBackendBaseUrl();

  if (!backendBaseUrl) {
    return {
      attempted: false,
      succeeded: false,
      status: 0,
      data: null,
      error: "KINGSMEN_BACKEND_BASE_URL is missing."
    };
  }

  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), 15000);

  try {
    const backendResponse = await fetch(`${backendBaseUrl}${endpoint}`, {
      method: "POST",
      headers: {
        "content-type": "application/json"
      },
      body: JSON.stringify(payload),
      signal: controller.signal
    });

    const data = (await backendResponse.json().catch(() => null)) as Record<string, unknown> | null;

    return {
      attempted: true,
      succeeded: backendResponse.status === 202 && data?.accepted === true,
      status: backendResponse.status,
      data,
      error: null
    };
  } catch (error) {
    return {
      attempted: true,
      succeeded: false,
      status: 0,
      data: null,
      error: error instanceof Error ? error.message : "Unknown backend bridge error."
    };
  } finally {
    clearTimeout(timer);
  }
}

}

        nextAction: accepted

        issues: accepted

    nextAction: "Controlled local Next API accepted the safe demo Free Destiny Map payload.",
    issues: []
  };

  if (localPersistenceEnabled) {
    responseBody.localWriteSucceeded = writeNextLocalProofRecord(payload, responseBody);
  }

function getFreePort() {

    server.once("error", reject);

async function fetchWithTimeout(url, options = {}, ms = 7000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), ms);
  try {
    return await fetch(url, { ...options, signal: controller.signal });
  } finally {
    clearTimeout(timer);
  }

async function waitForHttpOk(url, maxMs) {
  const started = Date.now();
  let lastError = "";
  while (Date.now() - started < maxMs) {
    try {
      const response = await fetchWithTimeout(url, {}, 7000);
      const text = await response.text();
      if (response.status === 200) {
        return { ok: true, status: response.status, text, lastError };
      }
      lastError = `status=${response.status}`;
    } catch (error) {
      lastError = error instanceof Error ? error.message : String(error);
    }
    await sleep(1000);
  }
  return { ok: false, status: 0, text: "", lastError };
}

async function stopProcessTree(child) {
  if (!child || !child.pid) return false;

  if (process.platform === "win32") {
    await new Promise((resolve) => {
      const killer = spawn("taskkill", ["/pid", String(child.pid), "/t", "/f"], {
        stdio: "ignore"
      });
      killer.once("close", () => resolve());
      killer.once("error", () => resolve());
    });
  } else {
    try {
      child.kill("SIGTERM");
    } catch {}
    await sleep(1500);
    if (child.exitCode === null) {
      try {
        child.kill("SIGKILL");
      } catch {}
    }
  }

  await sleep(1500);
  return true;
}
