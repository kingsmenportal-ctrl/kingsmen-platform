# BATCH 06 - Intake Persistence / Local Runtime Shell

## Purpose

This batch creates the first safe local persistence boundary for Free Destiny Map intake.

It connects the BATCH 05 intake service to a local runtime repository shell without claiming production storage, live database persistence, email delivery, payment readiness, or official application submission.

## What this batch creates

- local persistence contract
- local repository shell
- safe stored intake summary model
- audit event shell
- API-safe persistence service
- runtime route shell
- local data ignore rule
- validator and proof report

## What this batch does not do yet

- no production database write
- no Supabase service role usage
- no payment processing
- no email sending
- no document upload
- no official application submission
- no production runtime claim

## Runtime safety

Local persistence must be opt-in for development/runtime proof only.

Candidate production data must not be committed to Git.

The repository may create local JSONL records only when the runtime flag allows it. The committed repo includes a local data ignore rule so generated intake files are not tracked.

## Oracle Lite impact

none_runtime_yet.

This batch creates code and validation only. It does not start Docker, PostgreSQL, Redis, Nginx, n8n, Playwright, email, payment, or Oracle services.

## Next action

After validation, BATCH 07 should wire the public form shell to the intake endpoint or create the local API boot/run proof.

import type {
  FreeDestinyMapIntakePayload,
  FreeDestinyMapIntakeResult
} from "./free-destiny-map-intake";

export type FreeDestinyMapLocalStatus =
  | "LOCAL_SHELL_RECEIVED"
  | "LOCAL_SHELL_ACCEPTED"
  | "LOCAL_SHELL_REJECTED"
  | "LOCAL_SHELL_WRITE_DISABLED";

export interface FreeDestinyMapAuditEvent {
  eventType: string;
  message: string;
  occurredAt: string;
  productionReady: false;
}

export interface StoredFreeDestinyMapIntakeSummary {
  reference: string;
  status: FreeDestinyMapLocalStatus;
  createdAt: string;
  productionReady: false;
  storesProductionData: false;
  payload: FreeDestinyMapIntakePayload;
  result: FreeDestinyMapIntakeResult;
  auditEvents: FreeDestinyMapAuditEvent[];
}

export interface FreeDestinyMapPersistenceResult {
  accepted: boolean;
  reference: string;
  productionReady: false;
  localPersistenceEnabled: boolean;
  localWriteAttempted: boolean;
  localWriteSucceeded: boolean;
  storageMode: "disabled" | "local_jsonl_shell";
  safeNotice: string;
  nextAction: string;
  issues: FreeDestinyMapIntakeResult["issues"];
}

export const FREE_DESTINY_MAP_LOCAL_PERSISTENCE_NOTICE =
  "Local persistence is a development/runtime shell only. It is not production database storage, not official submission, not payment, not email delivery, and not document upload.";

export function buildFreeDestinyMapAuditEvent(eventType: string, message: string): FreeDestinyMapAuditEvent {
  return {
    eventType,
    message,
    occurredAt: new Date().toISOString(),
    productionReady: false
  };
}

import fs from "node:fs/promises";
import path from "node:path";
import type { StoredFreeDestinyMapIntakeSummary } from "../../../../packages/contracts/src/free-destiny-map-persistence";

export interface FreeDestinyMapLocalRepositoryOptions {
  baseDirectory?: string;
}

export class FreeDestinyMapLocalRepository {
  private readonly baseDirectory: string;

  constructor(options: FreeDestinyMapLocalRepositoryOptions = {}) {
    this.baseDirectory = options.baseDirectory ?? path.join(process.cwd(), "data", "local", "free-destiny-map");
  }

  async appendIntake(summary: StoredFreeDestinyMapIntakeSummary): Promise<{ path: string; written: true }> {
    await fs.mkdir(this.baseDirectory, { recursive: true });

    const filePath = path.join(this.baseDirectory, "intakes.jsonl");
    const line = `${JSON.stringify(summary)}\n`;

    await fs.appendFile(filePath, line, "utf8");

    return {
      path: filePath,
      written: true
    };
  }
}

import {
  buildFreeDestinyMapIntakeResult,
  normalizeFreeDestinyMapIntakePayload
} from "./free-destiny-map-intake.service";
import { FreeDestinyMapLocalRepository } from "../repositories/free-destiny-map-local.repository";
import {
  FREE_DESTINY_MAP_LOCAL_PERSISTENCE_NOTICE,
  buildFreeDestinyMapAuditEvent,
  type FreeDestinyMapPersistenceResult,
  type StoredFreeDestinyMapIntakeSummary
} from "../../../../packages/contracts/src/free-destiny-map-persistence";

export interface FreeDestinyMapPersistenceOptions {
  enableLocalWrite?: boolean;
  repository?: FreeDestinyMapLocalRepository;
}

export async function persistFreeDestinyMapIntakeShell(
  input: unknown,
  options: FreeDestinyMapPersistenceOptions = {}
): Promise<FreeDestinyMapPersistenceResult> {
  const payload = normalizeFreeDestinyMapIntakePayload(input);
  const result = buildFreeDestinyMapIntakeResult(payload);

  const localPersistenceEnabled = options.enableLocalWrite === true;
  const repository = options.repository ?? new FreeDestinyMapLocalRepository();

  if (!localPersistenceEnabled) {
    return {
      accepted: result.accepted,
      reference: result.reference,
      productionReady: false,
      localPersistenceEnabled: false,
      localWriteAttempted: false,
      localWriteSucceeded: false,
      storageMode: "disabled",
      safeNotice: FREE_DESTINY_MAP_LOCAL_PERSISTENCE_NOTICE,
      issues: result.issues,
      nextAction: "Local write is disabled. Set KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE=true only for controlled local runtime proof."
    };
  }

  const storedSummary: StoredFreeDestinyMapIntakeSummary = {
    reference: result.reference,
    status: result.accepted ? "LOCAL_SHELL_ACCEPTED" : "LOCAL_SHELL_REJECTED",
    createdAt: new Date().toISOString(),
    productionReady: false,
    storesProductionData: false,
    payload,
    result,
    auditEvents: [
      buildFreeDestinyMapAuditEvent("FREE_DESTINY_MAP_LOCAL_SHELL_RECEIVED", "Free Destiny Map intake reached local runtime shell."),
      buildFreeDestinyMapAuditEvent("FREE_DESTINY_MAP_LOCAL_SHELL_VALIDATED", result.accepted ? "Shell validation accepted intake." : "Shell validation returned issues.")
    ]
  };

  await repository.appendIntake(storedSummary);

  return {
    accepted: result.accepted,
    reference: result.reference,
    productionReady: false,
    localPersistenceEnabled: true,
    localWriteAttempted: true,
    localWriteSucceeded: true,
    storageMode: "local_jsonl_shell",
    safeNotice: FREE_DESTINY_MAP_LOCAL_PERSISTENCE_NOTICE,
    issues: result.issues,
    nextAction: result.accepted
      ? "Local shell wrote intake summary. Later batches must move this to governed database persistence."
      : "Local shell wrote rejected/issue summary for proof. Fix issues before progressing."
  };
}

import { persistFreeDestinyMapIntakeShell } from "../services/free-destiny-map-persistence.service";

type RouteLikeApp = {
  post: (
    path: string,
    handler: (
      request: { body?: unknown },
      reply: { code: (status: number) => { send: (payload: unknown) => unknown } }
    ) => unknown | Promise<unknown>
  ) => unknown;
};

export async function registerFreeDestinyMapPersistenceRoutes(app: RouteLikeApp) {
  app.post("/api/public/free-destiny-map/intake/local", async (request, reply) => {
    const result = await persistFreeDestinyMapIntakeShell(request.body, {
      enableLocalWrite: process.env.KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE === "true"
    });

    return reply.code(result.accepted ? 202 : 400).send({
      ...result,
      officialApplicationSubmitted: false,
      paymentInstructionSent: false,
      documentUploadAccepted: false,
      emailSent: false,
      productionReady: false
    });
  });
}

#!/usr/bin/env node
import fs from "node:fs";
import path from "node:path";

const root = process.cwd();
const reportPath = path.join(root, "reports/validation/free-destiny-map-persistence-latest.json");

const requiredFiles = [
  "docs/intake/free-destiny-map-local-runtime-shell.md",
  "manifests/intake/free-destiny-map-persistence-shell.json",
  "packages/contracts/src/free-destiny-map-persistence.ts",
  "apps/api/src/repositories/free-destiny-map-local.repository.ts",
  "apps/api/src/services/free-destiny-map-persistence.service.ts",
  "apps/api/src/routes/free-destiny-map-persistence.routes.ts",
  "data/local/.gitignore",
  "scripts/validators/validate-free-destiny-map-persistence.mjs",
  "reports/validation/free-destiny-map-persistence-latest.json"
];

function exists(rel) {
  return fs.existsSync(path.join(root, rel));
}

function read(rel) {
  return fs.readFileSync(path.join(root, rel), "utf8").replace(/^\uFEFF/, "");
}

function parseJson(rel) {
  return JSON.parse(read(rel));
}

const failures = [];
const warnings = [];
const checked = [];

for (const file of requiredFiles) {
  checked.push(file);
  if (!exists(file)) failures.push(`Missing required file: ${file}`);
  else if (fs.statSync(path.join(root, file)).size === 0) failures.push(`Empty required file: ${file}`);
}

if (exists("manifests/intake/free-destiny-map-persistence-shell.json")) {
  const manifest = parseJson("manifests/intake/free-destiny-map-persistence-shell.json");
  if (manifest.productionReady !== false) failures.push("Manifest must keep productionReady false.");
  if (manifest.storesProductionData !== false) failures.push("Manifest must declare storesProductionData false.");
  if (manifest.storesLocalShellData !== true) failures.push("Manifest must declare storesLocalShellData true.");
  if (manifest.takesPayment !== false) failures.push("Manifest must declare takesPayment false.");
  if (manifest.sendsEmail !== false) failures.push("Manifest must declare sendsEmail false.");
  if (manifest.collectsDocuments !== false) failures.push("Manifest must declare collectsDocuments false.");
  if (manifest.endpoint !== "/api/public/free-destiny-map/intake/local") failures.push("Manifest endpoint mismatch.");
  if (!String(manifest.oracleLiteImpact ?? "").includes("none_runtime_yet")) failures.push("Manifest must declare none_runtime_yet.");
}

const tokenChecks = [
  ["packages/contracts/src/free-destiny-map-persistence.ts", ["StoredFreeDestinyMapIntakeSummary", "FreeDestinyMapPersistenceResult", "FREE_DESTINY_MAP_LOCAL_PERSISTENCE_NOTICE", "storesProductionData: false"]],
  ["apps/api/src/repositories/free-destiny-map-local.repository.ts", ["FreeDestinyMapLocalRepository", "appendIntake", "intakes.jsonl", "fs.appendFile"]],
  ["apps/api/src/services/free-destiny-map-persistence.service.ts", ["persistFreeDestinyMapIntakeShell", "KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE", "localPersistenceEnabled", "storesProductionData: false"]],
  ["apps/api/src/routes/free-destiny-map-persistence.routes.ts", ["/api/public/free-destiny-map/intake/local", "KINGSMEN_ENABLE_LOCAL_INTAKE_WRITE", "officialApplicationSubmitted: false", "paymentInstructionSent: false", "documentUploadAccepted: false", "emailSent: false", "productionReady: false"]],
  ["data/local/.gitignore", ["*", "!.gitignore"]]
];

for (const [file, tokens] of tokenChecks) {
  if (!exists(file)) continue;
  const content = read(file);
  for (const token of tokens) {
    if (!content.includes(token)) failures.push(`${file} missing token: ${token}`);
  }
}

const forbiddenRuntimePatterns = [
  [/\bidentityNumber\s*[:=]/i, "identity number field assignment"],
  [/\bidNumber\s*[:=]/i, "id number field assignment"],
  [/\bpassportNumber\s*[:=]/i, "passport number field assignment"],
  [/\bpassword\s*[:=]/i, "password field assignment"],
  [/\bcardNumber\s*[:=]/i, "card number field assignment"],
  [/\bcvv\s*[:=]/i, "cvv field assignment"],
  [/\bdocumentBase64\s*[:=]/i, "document payload assignment"],
  [/officialApplicationSubmitted:\s*true/i, "official application submitted true"],
  [/paymentInstructionSent:\s*true/i, "payment instruction sent true"],
  [/emailSent:\s*true/i, "email sent true"],
  [/productionReady\s*[:=]\s*true/i, "productionReady true"]
];

const filesToScan = requiredFiles
  .filter((item) => exists(item))
  .filter((item) => !item.includes("validate-free-destiny-map-persistence.mjs"));

for (const file of filesToScan) {
  const content = read(file);
  for (const [pattern, label] of forbiddenRuntimePatterns) {
    if (pattern.test(content)) failures.push(`${file} contains forbidden runtime pattern: ${label}`);
  }
}

const passed = failures.length === 0;

const report = {
  schema: "kingsmen.validation-report.v1",
  module: "BATCH 06 - Intake Persistence / Local Runtime Shell",
  status: passed ? "VALIDATED" : "BUILT_PENDING_VALIDATION",
  passed,
  generated_at: new Date().toISOString(),
  files_expected: requiredFiles,
  files_checked: checked.length,
  missing_files: requiredFiles.filter((file) => !exists(file)),
  failed_checks: failures,
  warnings,
  oracle_lite_impact: "none_runtime_yet",
  control_plane_impact: "Adds local Free Destiny Map intake persistence boundary for runtime proof.",
  next_action: passed ? "Proceed to BATCH 07 - Public form to intake API wiring or local API boot proof." : "Repair Free Destiny Map persistence failures and rerun validator."
};

fs.mkdirSync(path.dirname(reportPath), { recursive: true });
fs.writeFileSync(reportPath, `${JSON.stringify(report, null, 2)}\n`, "utf8");

console.log(JSON.stringify(report, null, 2));
if (!passed) process.exit(1);
