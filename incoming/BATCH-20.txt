# BATCH 20 - Controlled Docker Local Build Proof

## Purpose

This batch performs the first controlled Docker local build proof for the Kingsmen Oracle Lite web and backend runtime boundary.

BATCH 19 prepared Docker and Oracle Lite local runtime files.

BATCH 20 checks whether Docker is installed and available. If Docker is available, this batch builds the local API and web images. If Docker is not installed or not running, it writes an honest unavailable proof report instead of pretending the build passed.

## What this batch may prove

- BATCH 19 container preparation exists and passed
- Docker CLI is installed
- Docker daemon is available
- API Oracle Lite image can build locally
- Web Oracle Lite image can build locally
- built images can be inspected
- no container process is started
- no Compose startup is performed
- no Oracle deployment is performed
- production/action flags remain false

## What this batch does not do

- no container process start
- no Compose startup
- no Oracle deployment
- no public production URL
- no PostgreSQL runtime
- no Redis queue
- no n8n runtime
- no Playwright runtime
- no email sending
- no payment processing
- no document upload
- no official application submission

## Honest unavailable mode

If Docker is missing or Docker Desktop is not running, this batch is allowed to produce:

`DOCKER_LOCAL_BUILD_UNAVAILABLE`

That is a valid environment proof, but it is not a successful image build. In that case, do not proceed to Docker runtime proof until Docker is installed and running.

## Oracle Lite impact

Controlled local image build proof only.

No Oracle deployment is performed.

## Next action

If image build is proven, BATCH 21 can run a controlled Docker local runtime proof.

If Docker is unavailable, start or install Docker first, then rerun BATCH 20.

function exists(rel) {

function read(rel) {

function check(name, passed, detail = "") {

function trimLog(lines, limit = 180) {
  return lines.join("").split(/\r?\n/).filter(Boolean).slice(-limit);
}

function safeEnv() {
  const env = {};
  const keys = [
    "PATH",
    "Path",
    "PATHEXT",
    "SystemRoot",
    "WINDIR",
    "TEMP",
    "TMP",
    "USERPROFILE",
    "APPDATA",
    "LOCALAPPDATA",
    "COMSPEC",
    "HOMEDRIVE",
    "HOMEPATH",
    "PROCESSOR_ARCHITECTURE",
    "PROCESSOR_IDENTIFIER",
    "NUMBER_OF_PROCESSORS",
    "DOCKER_HOST",
    "DOCKER_CONTEXT"
  ];

  for (const key of keys) {
    const value = process.env[key];
    if (!key || key.includes("=") || key.includes("\u0000")) continue;
    if (typeof value === "undefined" || String(value).includes("\u0000")) continue;
    env[key] = String(value);
  }

  env.CI = "1";
  env.NEXT_TELEMETRY_DISABLED = "1";

  return env;
}

function runDocker(args, options = {}) {
  const timeout = options.timeoutMs ?? 30000;
  const collectOutput = options.collectOutput !== false;

  return new Promise((resolve) => {
    const out = [];
    const err = [];
    let timedOut = false;

    const child = spawn("docker", args, {
      cwd: root,
      env: safeEnv(),
      stdio: ["ignore", "pipe", "pipe"],
      windowsHide: true
    });

    const timer = setTimeout(() => {
      timedOut = true;
      try {
        child.kill("SIGTERM");
      } catch {}
      if (process.platform === "win32" && child.pid) {
        const killer = spawn("taskkill", ["/pid", String(child.pid), "/t", "/f"], { stdio: "ignore" });
        killer.once("close", () => {});
        killer.once("error", () => {});
      }
    }, timeout);

    child.stdout.on("data", (chunk) => {
      const text = chunk.toString();
      out.push(text);
      if (collectOutput) stdoutLines.push(text);
    });

    child.stderr.on("data", (chunk) => {
      const text = chunk.toString();
      err.push(text);
      if (collectOutput) stderrLines.push(text);
    });

    child.once("error", (error) => {
      clearTimeout(timer);
      resolve({
        code: null,
        timedOut,
        stdout: out.join(""),
        stderr: err.join(""),
        error: error instanceof Error ? error.message : String(error)
      });
    });

    child.once("close", (code) => {
      clearTimeout(timer);
      resolve({
        code,
        timedOut,
        stdout: out.join(""),
        stderr: err.join(""),
        error: null
      });
    });
  });
}

  if (!allowedStatuses.includes(proof.status)) failures.push("Proof status must be DOCKER_LOCAL_BUILD_PROVEN or DOCKER_LOCAL_BUILD_UNAVAILABLE.");

  if (proof.composeUpInThisBatch !== false) failures.push("Proof must keep composeUpInThisBatch false.");
  if (proof.oracle_lite_impact !== "controlled_local_docker_build_only") failures.push("Proof must declare controlled_local_docker_build_only.");
  if (!Array.isArray(proof.checks) || proof.checks.length < 10) failures.push("Proof must include detailed checks.");

  if (proof.status === "DOCKER_LOCAL_BUILD_PROVEN") {
    if (proof.dockerAvailable !== true) failures.push("Build-proven status requires dockerAvailable true.");
    if (proof.dockerBuildsAttempted !== true) failures.push("Build-proven status requires dockerBuildsAttempted true.");
    if (proof.dockerBuildsSucceeded !== true) failures.push("Build-proven status requires dockerBuildsSucceeded true.");
    if (proof.apiBuildExitCode !== 0) failures.push("Build-proven status requires apiBuildExitCode 0.");
    if (proof.webBuildExitCode !== 0) failures.push("Build-proven status requires webBuildExitCode 0.");
    if (!proof.apiImageId) failures.push("Build-proven status requires apiImageId.");
    if (!proof.webImageId) failures.push("Build-proven status requires webImageId.");
  }

  if (proof.status === "DOCKER_LOCAL_BUILD_UNAVAILABLE") {
    warnings.push("Docker is unavailable. BATCH 20 is honestly validated as environment-blocked, not image-build-proven.");
    if (proof.dockerAvailable !== false) failures.push("Unavailable status requires dockerAvailable false.");
    if (proof.dockerBuildsAttempted !== false) failures.push("Unavailable status requires dockerBuildsAttempted false.");
    if (proof.dockerBuildsSucceeded !== false) failures.push("Unavailable status requires dockerBuildsSucceeded false.");
    if (!proof.unavailableReason) failures.push("Unavailable status requires unavailableReason.");
  }
}
