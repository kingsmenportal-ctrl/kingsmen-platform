# BATCH 13 - Repair Real Web Build Blockers / Add Next App Package Runtime

## Purpose

This batch repairs the real web boot blockers found by BATCH 12.

BATCH 12 did the correct thing: it refused to claim a real local web boot because the repo had no build script, no dev script, no Next dependency, no React dependency, no React DOM dependency, and no node_modules install evidence.

BATCH 13 adds the missing web runtime foundation so the next batch can attempt a controlled local web boot proof.

## What this batch creates or repairs

- `apps/web/package.json`
- web `dev`, `build`, and `start` scripts
- Next, React, React DOM, TypeScript, and type dependencies
- `apps/web/.gitignore`
- `apps/web/next.config.mjs`
- `apps/web/tsconfig.json`
- `apps/web/next-env.d.ts`
- `apps/web/src/app/layout.tsx`
- import-path repair for the Free Destiny Map form component if needed
- dependency install command
- runtime repair proof report
- validator report

## What this batch does not claim

- no production deployment
- no Oracle runtime
- no Docker Compose boot
- no PostgreSQL database
- no Redis queue
- no production web build claim yet
- no long-running dev server claim yet
- no email sending
- no payment
- no document upload
- no official application submission

## Oracle Lite impact

none_runtime_yet.

This only repairs the local web runtime foundation.

## Next action

After validation, rerun BATCH 12 readiness. If BATCH 12 changes to `READY_FOR_CONTROLLED_LOCAL_BOOT`, BATCH 14 can attempt a controlled local Next/dev-server boot proof with explicit timeout and shutdown.

#requires -Version 5.1
[CmdletBinding()]
param(
  [switch]$RunInstall
)

$ErrorActionPreference = "Stop"

$repoRoot = Get-Location
$webDir = Join-Path $repoRoot "apps\web"
$componentPath = Join-Path $repoRoot "apps\web\src\components\kingsmen\FreeDestinyMapApiForm.tsx"
$proofScript = Join-Path $repoRoot "scripts\proof\prove-free-destiny-map-real-web-runtime-repair.mjs"

if (-not (Test-Path $webDir -PathType Container)) {
  throw "Missing web directory: $webDir"
}

if (-not (Test-Path $proofScript -PathType Leaf)) {
  throw "Missing proof script: $proofScript"
}

if (Test-Path $componentPath -PathType Leaf) {
  $content = Get-Content $componentPath -Raw
  $updated = $content `
    -replace '\.\./\.\.//lib/kingsmen/free-destiny-map-client', '../../lib/kingsmen/free-destiny-map-client' `
    -replace '\.\./\.\./\.\./\.\./packages/contracts/src/free-destiny-map-api-wiring', '../../../../../packages/contracts/src/free-destiny-map-api-wiring'

  if ($updated -ne $content) {
    $utf8 = New-Object System.Text.UTF8Encoding($false)
    [System.IO.File]::WriteAllText($componentPath, $updated, $utf8)
    Write-Host "Repaired FreeDestinyMapApiForm import paths."
  }
}

if ($RunInstall) {
  Write-Host "Installing apps/web dependencies with npm."
  Push-Location $webDir
  try {
    $npmCommand = Get-Command npm.cmd -ErrorAction SilentlyContinue
    if ($null -eq $npmCommand) {
      $npmCommand = Get-Command npm -ErrorAction Stop
    }

    & $npmCommand.Source install --no-audit --no-fund
    if ($LASTEXITCODE -ne 0) {
      throw "npm install failed with exit code $LASTEXITCODE"
    }
  } finally {
    Pop-Location
  }
} else {
  Write-Host "Skipping npm install. Use -RunInstall to repair dependency-install-evidence-present."
}

Write-Host "Kingsmen Free Destiny Map real web runtime repair proof"
Write-Host "This repairs local web runtime readiness. It does not claim production, Oracle runtime, email, payment, documents, or official submissions."

node $proofScript

if ($LASTEXITCODE -ne 0) {
  throw "Free Destiny Map web runtime repair proof failed."
}

function exists(rel) {
  return fs.existsSync(path.join(root, rel));

function read(rel) {
  return fs.readFileSync(path.join(root, rel), "utf8").replace(/^\uFEFF/, "");
}

function parseJson(rel) {
  return JSON.parse(read(rel));
}

function check(name, passed, detail = "") {
  checks.push({ name, passed, detail });
}

const webPackagePath = "apps/web/package.json";

  check("react-types-declared", Boolean(devDeps["@types/react"]), devDeps["@types/react"] ?? "missing");
}

if (exists("apps/web/.gitignore")) {
  const ignore = read("apps/web/.gitignore");
  check("web-gitignore-blocks-node-modules", ignore.includes("node_modules"), "node_modules ignored");
  check("web-gitignore-blocks-next-output", ignore.includes(".next"), ".next ignored");
}

if (exists("apps/web/src/app/layout.tsx")) {
  const layout = read("apps/web/src/app/layout.tsx");

  check("form-still-has-consent-fields", form.includes("consentToBeContacted") && form.includes("consentToProcessBasicInfo"), "consent fields present");
}

const installEvidence = exists("apps/web/node_modules/next/package.json") &&
  exists("apps/web/node_modules/react/package.json") &&

check("dependency-install-evidence-present", installEvidence, installEvidence ? "apps/web/node_modules contains next/react/react-dom" : "Run npm install --prefix apps/web --no-audit --no-fund");

if (exists("apps/web/package-lock.json")) {
  check("web-package-lock-created", true, "apps/web/package-lock.json");

const forbiddenFiles = [

import fs from "node:fs";
import path from "node:path";

const root = process.cwd();
const reportPath = path.join(root, "reports/validation/free-destiny-map-real-web-runtime-repair-latest.json");

const requiredFiles = [
  "docs/runtime/free-destiny-map-real-web-runtime-repair.md",

function exists(rel) {
  return fs.existsSync(path.join(root, rel));

function read(rel) {
  return fs.readFileSync(path.join(root, rel), "utf8").replace(/^\uFEFF/, "");

function parseJson(rel) {
  return JSON.parse(read(rel));
}

const failures = [];
const warnings = [];
const checked = [];

for (const file of requiredFiles) {
  checked.push(file);
  if (!exists(file)) failures.push(`Missing required file: ${file}`);
  else if (fs.statSync(path.join(root, file)).size === 0) failures.push(`Empty required file: ${file}`);
}

if (exists("manifests/runtime/free-destiny-map-real-web-runtime-repair.json")) {
  const manifest = parseJson("manifests/runtime/free-destiny-map-real-web-runtime-repair.json");
  if (manifest.productionReady !== false) failures.push("Manifest must keep productionReady false.");
  if (manifest.storesProductionData !== false) failures.push("Manifest must declare storesProductionData false.");
  if (manifest.takesPayment !== false) failures.push("Manifest must declare takesPayment false.");
  if (manifest.sendsEmail !== false) failures.push("Manifest must declare sendsEmail false.");
  if (manifest.collectsDocuments !== false) failures.push("Manifest must declare collectsDocuments false.");
  if (manifest.submitsOfficialApplications !== false) failures.push("Manifest must declare submitsOfficialApplications false.");
  if (!String(manifest.oracleLiteImpact ?? "").includes("none_runtime_yet")) failures.push("Manifest must declare none_runtime_yet.");

  if (!pkg.dependencies?.react) failures.push("apps/web package must declare react dependency.");
  if (!pkg.dependencies?.["react-dom"]) failures.push("apps/web package must declare react-dom dependency.");
  if (!pkg.devDependencies?.typescript) failures.push("apps/web package must declare typescript devDependency.");
}

const tokenChecks = [
  ["apps/web/.gitignore", ["node_modules", ".next"]],
  ["apps/web/next.config.mjs", ["reactStrictMode", "externalDir"]],
  ["apps/web/tsconfig.json", ["@kingsmen/contracts", "jsx"]],
  ["apps/web/src/app/layout.tsx", ["RootLayout", "Kingsmen"]],
  ["packages/contracts/src/free-destiny-map-real-web-runtime-repair.ts", ["WEB_RUNTIME_REPAIR_APPLIED", "canAttemptNextControlledBoot"]],

if (!exists("apps/web/node_modules/react-dom/package.json")) failures.push("apps/web/node_modules/react-dom/package.json must exist after npm install.");

  next_action: passed ? "Rerun BATCH 12 readiness. If ready, proceed to BATCH 14 controlled local web boot proof." : "Repair runtime package blockers and rerun proof plus validator."
};

fs.mkdirSync(path.dirname(reportPath), { recursive: true });
fs.writeFileSync(reportPath, `${JSON.stringify(report, null, 2)}\n`, "utf8");

console.log(JSON.stringify(report, null, 2));
if (!passed) process.exit(1);
